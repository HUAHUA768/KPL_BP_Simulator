│   │   │   ├── useWebSocket.ts

│   │   ├── types/                # TypeScript 类型定义

├── backend/                      # 后端（Go + Gin）

│   │   └── server/
│   │       └── main.go           # 入口，启动 HTTP & WebSocket 服务
│   ├── internal/

│   │   └── websocket/            # WebSocket 管理
│   │       ├── hub.go            # 房间管理器

└── docs/
